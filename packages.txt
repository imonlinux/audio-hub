# Audio Hub required packages (Debian / Raspberry Pi OS Trixie, aarch64)
#
# Unversioned on purpose: pins (pkg=ver) break installs as soon as a repo
# updates. Reference build was pipewire 1.4.2 / wireplumber 0.5.8 /
# bluez 5.82 — anything >= these from the Trixie repos is expected to work.

# Audio stack
pipewire
pipewire-audio
pipewire-alsa
pipewire-pulse
pipewire-bin
wireplumber
libspa-0.2-bluetooth
pulseaudio-utils

# Bluetooth
bluez
bluez-firmware
bluez-obexd
bluez-tools   # also provides /usr/bin/bt-agent (there is no separate bt-agent package)
rfkill

# Python dependencies for hubd (system packages, no venv needed)
python3-evdev
python3-paho-mqtt
python3-pulsectl

# Network / system
network-manager
iw
usbutils
curl
